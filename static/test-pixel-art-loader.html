            📝 Utilisation du plugin
            
const loader = new PixelArtLoader({
    containerId: 'pixel-art-mac',
    files: ['mac_1.txt', 'mac_2.txt', 'mac_3.txt'],
    interval: 8080,     // Rotation toutes les 5 secondes
    lineDelay: 50       // 50ms entre chaque ligne
});

loader.start();

        


        

            🎯 Méthodes disponibles
            
    
                
  • loader.start() - Démarre l'animation
    • 
                
  • loader.stop() - Arrête l'animation
    • 
                
  • loader.next() - Image suivante
    • 
                
  • loader.previous() - Image précédente
    • 
                
  • loader.reload() - Recharge les fichiers
    • 
                
  • loader.showArtByIndex(index) - Affiche une image spécifique
    • 
                
  • loader.destroy() - Détruit le plugin
